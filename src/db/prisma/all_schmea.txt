// ============================================================================
//  Scholar — consolidated Prisma schema (single file)
//
//  This is a read-only snapshot of the project's multi-file Prisma schema.
//  The source of truth is the directory `src/db/prisma/`:
//
//      src/db/prisma/schema.prisma     datasource + generator blocks
//      src/db/prisma/enums/*.prisma    15 enums
//      src/db/prisma/schemas/*.prisma  25 models
//
//  `prisma.config.ts` points Prisma at the directory (`schema: "./src/db/prisma"`),
//  so every `.prisma` file inside it is loaded as part of one schema. This file is
//  named `.txt` on purpose so Prisma does not try to parse it and the live
//  multi-file schema stays untouched.
//
//  Contents below are copied verbatim. Regenerate by re-concatenating enums/
//  and schemas/ — do not edit by hand.
//
//  ENUMS:  applicationStatus, feedbackCategory, jobType, licenseType,
//          presenceStatus, publicationAccessType, publicationIndex, publicationType,
//          Publisher, quartile, seniorityLevel, subBucket, teamCategory,
//          userRole, workflow
//  MODELS: academicPublications, academicResearchers, accounts, companies,
//          companyBranches, departments, faculties, feedbacks,
//          graduationProjectApplications, graduationProjects, jobApplications,
//          jobs, profiles, publicationDomains, publicationEditoralStat,
//          publicationPricing, publicationSubcategories, publicationYearlyMetrics,
//          researchPapers, sessions, teamMembers, universities, universityStaff,
//          users, verifications
// ============================================================================

datasource db {
  provider = "postgresql"

  provider = "prisma-client-js"

// from: schemas/academicPublications.prisma
model AcademicPublication {
  id                 Int                     @id @default(autoincrement())
  subCategoryId      Int
  acronym            String?                 @db.VarChar(100)
  publisher          Publisher
  publicationType    PublicationType
  title              String
  issn               String?                 @unique @db.Char(8)
  eissn              String?                 @unique @db.Char(8)
  issnCdrom          String?                 @unique @db.Char(8)
  URL                String?                 @db.VarChar(500)
  yearLunched        Int?
  openAccessType     PublicationAccessType
  specificFocusScope String?                 @db.VarChar(500)
  workflow           Workflow
  licenseType        LicenseType?
  journalScope       String?                 @db.Text
  searchVector  Unsupported("tsvector")? @map("search_vector")
  imprint            String?
  subBucket          SubBucket?
  createdAt          DateTime                @default(now())
  updatedAt          DateTime                @updatedAt


  // Relations
  subCategory        PublicationSubCategory  @relation(fields: [subCategoryId], references: [id], onDelete: Cascade)
  yearlyMetrics      PublicationYearlyMetric[]
  editorialStats     PublicationEditorialStat[]
  pricings           PublicationPricing[]


  @@index([subCategoryId])
  @@index([title])
  @@index([acronym])
  @@index([openAccessType])
  @@index([licenseType])
  @@index([specificFocusScope])
  @@map("academicPublications")
}

// from: schemas/publicationYearlyMetrics.prisma
model PublicationYearlyMetric {
  id                        Int                 @id @default(autoincrement())
  publicationId             Int
  metricYear                Int
  indexingService           PublicationIndex?
  impactFactor              Decimal?            @db.Decimal(10, 4)
  impactFactor5yr           Decimal?            @db.Decimal(10, 4)
  quartile                  Quartile?
  jci                       Decimal?            @db.Decimal(10, 4)
  sjr                       Decimal?            @db.Decimal(10, 4)
  h5Index                   Decimal?            @db.Decimal(10, 4)
  eigenfactor               Decimal?            @db.Decimal(10, 6)
  articleInfluenceScore     Decimal?            @db.Decimal(10, 4)
  citescore                 Decimal?            @db.Decimal(10, 4)
  totalCitations            Int
  articleDownloads          Int
  createdAt                 DateTime            @default(now())
  updatedAt                 DateTime            @updatedAt

  // Relations
  publication               AcademicPublication @relation(fields: [publicationId], references: [id], onDelete: Cascade)

  // Composite UK with pub_id and metric_year
  @@unique([publicationId, metricYear, indexingService])
  @@index([metricYear, citescore(sort: Desc)])
  @@index([metricYear, impactFactor(sort: Desc)])
  @@index([metricYear, quartile])
  @@index([metricYear, sjr(sort: Desc)])
  @@map("publicationYearlyMetrics")
}
